{通达信-主图叠加-手机端缠论笔（纯通达信语言无法做出标准的缠论新/旧笔，凑合看吧）}
X_1:=TOPRANGE(HIGH);
X_2:=LOWRANGE(LOW);
X_3:=LLVBARS(HIGH,X_1);
X_4:=LLVBARS(LOW,X_1);
X_5:=HHVBARS(LOW,X_2);
X_6:=HHVBARS(HIGH,X_2);
X_7:=HHV(LOW,X_3+1)>LLV(HIGH,X_3+1);
X_8:=HHV(LOW,X_5+1)>LLV(HIGH,X_5+1);
X_9:=COUNT(LOW>=REF(LOW,1),X_4)>2 AND COUNT(HIGH>=REF(HIGH,1),X_3)>2;
X_10:=COUNT(HIGH<=REF(HIGH,1),X_6)>2 AND COUNT(LOW<=REF(LOW,1),X_5)>2;
X_11:=X_1 AND X_7 AND X_9 AND X_4>2; 
X_12:=X_2 AND X_8 AND X_10 AND X_6>2;
X_13:=BARSLAST(X_11);
X_14:=BARSLAST(X_12);
X_15:=IF(X_13=0,X_14,X_13)>=3;
X_16:=X_11 AND X_15 AND HIGH=HHV(HIGH,BARSLAST(X_12));
X_17:=X_12 AND X_15 AND LOW=LLV(LOW,BARSLAST(X_11));
上笔:DRAWLINE(X_17,LOW,X_16,HIGH,0),COLORGREEN,LINETHICK2;
下笔:DRAWLINE(X_16,HIGH,X_17,LOW,0),COLORGREEN,LINETHICK2;
